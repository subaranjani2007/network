#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define CRC_POLY 0x07

unsigned char compute_crc8(const char *bits) {
    unsigned char crc = 0x00;
    for (int i = 0; bits[i] != '\0'; i++) {
        int bit = bits[i] - '0';
        int msb = (crc >> 7) & 1;
        crc = (crc << 1) | bit;
        if (msb) {
            crc ^= CRC_POLY;
        }
    }
    for (int i = 0; i < 8; i++) {
        int msb = (crc >> 7) & 1;
        crc = crc << 1;
        if (msb) {
            crc ^= CRC_POLY;
        }
    }
    return crc;
}

void append_crc(const char *data, char *output) {
    unsigned char crc = compute_crc8(data);
    strcpy(output, data);
    int len = strlen(data);
    for (int i = 7; i >= 0; i--) {
        output[len + (7 - i)] = ((crc >> i) & 1) ? '1' : '0';
    }
    output[len + 8] = '\0';
}

void bit_stuff(const char *input, char *output) {
    int count = 0;
    int j = 0;
    for (int i = 0; input[i] != '\0'; i++) {
        output[j++] = input[i];
        if (input[i] == '1') {
            count++;
            if (count == 5) {
                output[j++] = '0';
                count = 0;
            }
        } else {
            count = 0;
        }
    }
    output[j] = '\0';
}

void bit_destuff(const char *input, char *output) {
    int count = 0;
    int j = 0;
    for (int i = 0; input[i] != '\0'; i++) {
        output[j++] = input[i];
        if (input[i] == '1') {
            count++;
            if (count == 5) {
                if (input[i + 1] == '0') {
                    i++;
                }
                count = 0;
            }
        } else {
            count = 0;
        }
    }
    output[j] = '\0';
}

int verify_crc8(const char *frame) {
    int len = strlen(frame);
    if (len < 8) return 0;
   
    char data_part[500];
    strncpy(data_part, frame, len - 8);
    data_part[len - 8] = '\0';
   
    unsigned char calc_crc = compute_crc8(data_part);
   
    unsigned char rec_crc = 0;
    for (int i = len - 8; i < len; i++) {
        rec_crc = (rec_crc << 1) | (frame[i] - '0');
    }
   
    return calc_crc == rec_crc;
}

void inject_error(const char *input, char *output, int error_type) {
    strcpy(output, input);
    int len = strlen(input);
    if (len == 0) return;

    if (error_type == 1) {
        int pos = len / 2;
        output[pos] = (output[pos] == '1') ? '0' : '1';
    } else if (error_type == 2) {
        int pos1 = len / 4;
        int pos2 = len / 2;
        if (pos1 < len) output[pos1] = (output[pos1] == '1') ? '0' : '1';
        if (pos2 < len) output[pos2] = (output[pos2] == '1') ? '0' : '1';
    }
}

int main() {
    char original_data[200];
    char data_with_crc[250];
    char stuffed_frame[300];
    char transmitted_frame[300];
    char received_frame[300];
    char destuffed_data[300];
    char final_data[200];
    int choice;

    printf("Enter original binary data sequence: ");
    scanf("%s", original_data);

    append_crc(original_data, data_with_crc);
    bit_stuff(data_with_crc, stuffed_frame);

    printf("\n--- Transmission Simulation Options ---");
    printf("\0121. No Error");
    printf("\0122. Single Bit Error");
    printf("\0123. Multiple Bit Error");
    printf("\012Enter choice (1-3): ");
    scanf("%d", &choice);

    if (choice == 2) {
        inject_error(stuffed_frame, transmitted_frame, 1);
    } else if (choice == 3) {
        inject_error(stuffed_frame, transmitted_frame, 2);
    } else {
        inject_error(stuffed_frame, transmitted_frame, 0);
    }

    strcpy(received_frame, transmitted_frame);
    bit_destuff(received_frame, destuffed_data);

    int is_valid = verify_crc8(destuffed_data);

    printf("\n============================================\n");
    printf("Original Data:        %s\n", original_data);
    printf("Data with CRC:        %s\n", data_with_crc);
    printf("Stuffed Frame:        %s\n", stuffed_frame);
    printf("Transmitted Frame:    %s\n", transmitted_frame);
    printf("Received Frame:       %s\n", received_frame);
    printf("Destuffed Data:       %s\n", destuffed_data);
   
    if (is_valid) {
        int final_len = strlen(destuffed_data) - 8;
        strncpy(final_data, destuffed_data, final_len);
        final_data[final_len] = '\0';
        printf("Error Detection:      SUCCESS (No errors detected)\n");
        printf("Accepted Data:        %s\n", final_data);
    } else {
        printf("Error Detection:      FAILURE (Errors detected in transmission!)\n");
    }
    printf("============================================\n");

    return 0;
}
