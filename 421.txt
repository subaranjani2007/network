#define HASH_SIZE 400003

typedef struct Node {
    int val;
    struct Node* next;
} Node;

typedef struct {
    Node* bucket[HASH_SIZE];
} HashSet;

void insert(HashSet* set, int val) {
    int key = (val < 0 ? -val : val) % HASH_SIZE;
    Node* curr = set->bucket[key];
    while (curr) {
        if (curr->val == val) return;
        curr = curr->next;
    }
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->val = val;
    newNode->next = set->bucket[key];
    set->bucket[key] = newNode;
}

int contains(HashSet* set, int val) {
    int key = (val < 0 ? -val : val) % HASH_SIZE;
    Node* curr = set->bucket[key];
    while (curr) {
        if (curr->val == val) return 1;
        curr = curr->next;
    }
    return 0;
}

void clearSet(HashSet* set) {
    for (int i = 0; i < HASH_SIZE; i++) {
        Node* curr = set->bucket[i];
        while (curr) {
            Node* temp = curr;
            curr = curr->next;
            free(temp);
        }
        set->bucket[i] = NULL;
    }
}

int findMaximumXOR(int* nums, int numsSize) {
    int maxResult = 0;
    int mask = 0;
    HashSet set;
    memset(&set, 0, sizeof(HashSet));

    for (int i = 30; i >= 0; i--) {
        mask |= (1 << i);
        int greedyTry = maxResult | (1 << i);

        for (int j = 0; j < numsSize; j++) {
            insert(&set, nums[j] & mask);
        }

        for (int j = 0; j < numsSize; j++) {
            int prefix = nums[j] & mask;
            if (contains(&set, greedyTry ^ prefix)) {
                maxResult = greedyTry;
                break;
            }
        }
        clearSet(&set);
    }

    return maxResult;
}
