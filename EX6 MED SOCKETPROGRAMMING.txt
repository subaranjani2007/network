SERVER.C
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <unistd.h>
#define PORT 8080

int is_palindrome(char *str) {
    int l = 0;
    int h = strlen(str) - 1;
    while (h > l && (str[h] == '\n' || str[h] == '\r' || str[h] == ' ')) {
        h--;
    }
    while (h > l) {
        if (str[l++] != str[h--]) return 0;
    }
    return 1;
}

int main(int argc, char const* argv[]) {
    int server_fd, new_socket;
    ssize_t valread;
    struct sockaddr_in address;
    int opt = 1;
    socklen_t addrlen = sizeof(address);
    char buffer[1024] = { 0 };

    if ((server_fd = socket(AF_INET, SOCK_STREAM, 0)) < 0) {
        perror("socket failed");
        exit(EXIT_FAILURE);
    }

    if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR | SO_REUSEPORT, &opt, sizeof(opt))) {
        perror("setsockopt");
        exit(EXIT_FAILURE);
    }
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);

    if (bind(server_fd, (struct sockaddr*)&address, sizeof(address)) < 0) {
        perror("bind failed");
        exit(EXIT_FAILURE);
    }
    if (listen(server_fd, 3) < 0) {
        perror("listen");
        exit(EXIT_FAILURE);
    }
    if ((new_socket = accept(server_fd, (struct sockaddr*)&address, &addrlen)) < 0) {
        perror("accept");
        exit(EXIT_FAILURE);
    }
  
    valread = read(new_socket, buffer, 1024 - 1); 
    printf("Received string from client: %s\n", buffer);

    char *result;
    if (is_palindrome(buffer)) {
        result = "The string IS a palindrome";
    } else {
        result = "The string IS NOT a palindrome";
    }

    send(new_socket, result, strlen(result), 0);
    printf("Result sent back to client\n");

    close(new_socket);
    close(server_fd);
    return 0;
}

#include <arpa/inet.h>
#include <stdio.h>
#include <string.h>
#include <sys/socket.h>
#include <unistd.h>
#define PORT 8080

int main(int argc, char const* argv[]) {
    int status, valread, client_fd;
    struct sockaddr_in serv_addr;
    char text_to_send[1024];
    char buffer[1024] = { 0 };

    if ((client_fd = socket(AF_INET, SOCK_STREAM, 0)) < 0) {
        printf("\n Socket creation error \n");
        return -1;
    }

    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(PORT);

    if (inet_pton(AF_INET, "127.0.0.1", &serv_addr.sin_addr) <= 0) {
        printf("\nInvalid address/ Address not supported \n");
        return -1;
    }

    if ((status = connect(client_fd, (struct sockaddr*)&serv_addr, sizeof(serv_addr))) < 0) {
        printf("\nConnection Failed \n");
        return -1;
    }
  
    printf("Enter a string to check for palindrome: ");
    fgets(text_to_send, sizeof(text_to_send), stdin);
    text_to_send[strcspn(text_to_send, "\n")] = 0;

    send(client_fd, text_to_send, strlen(text_to_send), 0);
    printf("String sent to server\n");

    valread = read(client_fd, buffer, 1024 - 1); 
    printf("Server response: %s\n", buffer);

    close(client_fd);
    return 0;
}
